/** * HAMI COFFEE - ORDER BACKEND * Gửi đơn đến Telegram * * Script Properties cần có: * TELEGRAM_BOT_TOKEN * TELEGRAM_CHAT_ID_1 * TELEGRAM_CHAT_ID_2 * ORDER_COUNTER */ function doGet() { return json_({ ok: true, service: "HAMI Coffee Order API" }); } function doPost(e) { try { const data = JSON.parse(e.postData.contents || "{}"); const props = PropertiesService.getScriptProperties(); const token = props.getProperty("TELEGRAM_BOT_TOKEN"); const chatIds = [ props.getProperty("TELEGRAM_CHAT_ID_1"), props.getProperty("TELEGRAM_CHAT_ID_2") ].filter(Boolean); if (!token) { throw new Error("Chưa cấu hình TELEGRAM_BOT_TOKEN."); } if (chatIds.length === 0) { throw new Error("Chưa cấu hình Telegram Chat ID."); } if (!data.name) { throw new Error("Chưa có tên khách hàng."); } if (!data.items || !data.items.length) { throw new Error("Chưa có món nào."); } // Tạo số đơn hàng const lock = LockService.getScriptLock(); lock.waitLock(10000); let orderNo; try { orderNo = nextOrderNo_(); } finally { lock.releaseLock(); } // Danh sách món const lines = data.items.map(function(x) { const itemName = escapeMd_(x.name); const itemSize = x.size ? ` (${escapeMd_(x.size)})` : ""; const qty = Number(x.qty || 1); // Món chưa có giá if (x.price === null || x.price === undefined || x.price === "") { return `• ${itemName}${itemSize} × ${qty} — Giá quán báo sau`; } // Món có giá const subTotal = Number(x.price) * qty; return `• ${itemName}${itemSize} × ${qty} — ${money_(subTotal)}`; }).join("\n"); // Tổng tiền let totalText; if (data.hasPendingPrice) { totalText = `${money_(data.total || 0)} + món chưa báo giá`; } else { totalText = money_(data.total || 0); } // Nội dung Telegram const text = `🔔 *ĐƠN HÀNG MỚI #${orderNo}* 👤 *Khách hàng:* ${escapeMd_(data.name)} 🏫 *Lớp / Số bàn:* ${escapeMd_(data.className || "Không có")} ━━━━━━━━━━━━━━ ☕ *MÓN ĐÃ CHỌN:* ${lines} ━━━━━━━━━━━━━━ 💰 *TỔNG TẠM TÍNH:* ${escapeMd_(totalText)} 📝 *Ghi chú:* ${escapeMd_(data.note || "Không có")} 🕒 *Thời gian:* ${escapeMd_(new Date().toLocaleString("vi-VN", { timeZone: "Asia/Ho_Chi_Minh" }))}`; const url = `https://api.telegram.org/bot${token}/sendMessage`; // Gửi đến tất cả Telegram Chat ID const results = chatIds.map(function(chatId) { const response = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", payload: JSON.stringify({ chat_id: chatId, text: text, parse_mode: "Markdown" }), muteHttpExceptions: true }); const telegram = JSON.parse(response.getContentText()); return { chatId: chatId, ok: telegram.ok, error: telegram.ok ? null : telegram.description }; }); const success = results.some(function(r) { return r.ok; }); if (!success) { throw new Error( results .map(function(r) { return r.error; }) .filter(Boolean) .join(" | ") || "Không gửi được thông báo Telegram." ); } return json_({ ok: true, orderNo: orderNo }); } catch (err) { return json_({ ok: false, error: String(err.message || err) }); } } function nextOrderNo_() { const props = PropertiesService.getScriptProperties(); const n = Number( props.getProperty("ORDER_COUNTER") || "0" ) + 1; props.setProperty( "ORDER_COUNTER", String(n) ); return String(n).padStart(3, "0"); } function money_(n) { return Number(n || 0) .toLocaleString("vi-VN") + "đ"; } function escapeMd_(s) { return String(s || "") .replace( /([_*[\]`])/g, "\\$1" ); } function json_(obj) { return ContentService .createTextOutput( JSON.stringify(obj) ) .setMimeType( ContentService.MimeType.JSON ); }
